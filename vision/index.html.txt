    <script>
        // Función interactiva para simular reservas locales al hacer clic en los botones
        function cambiarEstado(id) {
            const slot = document.getElementById(`slot-${id}`);
            const badge = slot.querySelector('.status-badge');
            const btn = slot.querySelector('button');

            if (badge.textContent === 'Disponible') {
                // Cambiar a Reservado
                slot.className = "p-4 rounded-xl border border-amber-500 bg-amber-950/30 flex flex-col items-center justify-between gap-3 transition-all duration-300";
                badge.className = "status-badge text-xs px-2.5 py-1 rounded-full bg-amber-500 text-black font-bold shadow";
                badge.textContent = "Reservado";
                btn.className = "w-full py-1.5 text-xs bg-amber-600 hover:bg-amber-500 text-white rounded-lg font-semibold transition shadow";
                btn.textContent = "Liberar";
            } else if (badge.textContent === 'Reservado') {
                // Cambiar a Disponible
                slot.className = "p-4 rounded-xl border border-emerald-500 bg-emerald-950/30 flex flex-col items-center justify-between gap-3 transition-all duration-300";
                badge.className = "status-badge text-xs px-2.5 py-1 rounded-full bg-emerald-500 text-black font-bold shadow";
                badge.textContent = "Disponible";
                btn.className = "w-full py-1.5 text-xs bg-emerald-600 hover:bg-emerald-500 text-white rounded-lg font-semibold transition shadow";
                btn.textContent = "Reservar";
            }
            // Nota: Los puestos "Ocupados" están bloqueados porque se gestionan automáticamente a través de la detección por visión artificial.
        }
    </script>
